- Uses `git grep` with explicit pathspec excludes (per 00-RESEARCH.md Open Question 4).

Body shape (lift from 00-RESEARCH.md Pattern 1 + Open Question 4 with the exclusion list applied):

```bash
#!/usr/bin/env bash
set -euo pipefail

# Forbidden-symbol grep gate — mirrors the lefthook pre-commit hooks.
# See CONTRIBUTING.md for the three rules these patterns enforce.

EXCLUDES=(
  ':(exclude).planning/'
  ':(exclude).legacy/'
  ':(exclude)CONTRIBUTING.md'
  ':(exclude)PROJECT.md'
  ':(exclude)README.md'
  ':(exclude).github/scripts/forbidden-symbols.sh'
  ':(exclude).github/workflows/ci.yml'
  ':(exclude)lefthook.yml'
)

PATTERNS=(
  'os\.ReadFile.*[Ss]chema'      # Rule 2: runtime schema interpretation
  'interface\s*\{\s*Path\('       # Rule 1: Property-tree relapse signature
  'Property\s+interface'          # Rule 1: interface declaration form
  'Entra|Okta|Azure\s*AD|AzureAD' # Rule 3: vendor-name accommodation
)

failed=0
for pat in "${PATTERNS[@]}"; do
  if git grep -nE "$pat" -- "${EXCLUDES[@]}" 2>/dev/null; then
    echo "::error::Forbidden pattern matched: $pat (see CONTRIBUTING.md)"
    failed=1
  fi
done

if [ "$failed" -ne 0 ]; then
  echo "Forbidden-symbol grep gate failed. See CONTRIBUTING.md for the rules."
  exit 1
fi

echo "Forbidden-symbol grep gate: OK"
```

Create `.github/workflows/ci.yml` with EXACTLY four jobs lifted from 00-RESEARCH.md Pattern 1, adapted to call the extracted script:

```yaml
name: ci
on:
  push:
    branches: [next, master, main]
  pull_request:

jobs:
  build-workspace:
    name: build (workspace, go ${{ matrix.go }})
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        go: ['1.25', '1.26']
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-go@v6
        with:
          go-version: ${{ matrix.go }}
      - name: Recreate workspace
        run: |
          go work init
          go work use ./gen ./rt
      - name: Build workspace
        run: go build ./...

  build-isolated:
    name: build (GOWORK=off, ${{ matrix.module }}, go ${{ matrix.go }})
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        go: ['1.25', '1.26']
        module: [gen, rt]
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-go@v6
        with:
          go-version: ${{ matrix.go }}
      - name: Build module in isolation
        env:
          GOWORK: 'off'
        run: |
          cd ${{ matrix.module }}
          go build ./...

  forbidden-symbols:
    name: forbidden-symbol grep gate
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - name: Grep for relapse signatures
        run: ./.github/scripts/forbidden-symbols.sh

  commit-msg-lint:
    name: commit message vendor-name gate
    runs-on: ubuntu-latest
    if: github.event_name == 'pull_request'
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - name: Check PR commit messages for vendor names
        run: |
          base=${{ github.event.pull_request.base.sha }}
          head=${{ github.event.pull_request.head.sha }}
          if git log "${base}..${head}" --format=%B | grep -iE 'Entra|Okta|Azure\s*AD|AzureAD'; then
            echo "::error::Commit message contains vendor name (per CONTRIBUTING.md Rule 3)."
            exit 1
          fi
```
